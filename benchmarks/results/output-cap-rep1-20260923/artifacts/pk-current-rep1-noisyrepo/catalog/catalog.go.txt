package catalog

import "strings"

// Source maps a request path to one enabled handler.
type Source struct {
	Name     string
	Prefix   string
	Priority int
	Enabled  bool
}

// Select chooses the best enabled source for path.
func Select(sources []Source, path string) (Source, bool) {
	var best Source
	found := false

	for _, source := range sources {
		prefix := source.Prefix
		if !source.Enabled || prefix == "" || !strings.HasPrefix(path, prefix) {
			continue
		}

		// A slash prefix matches any absolute path. Other prefixes must
		// match the complete path or end at a path-component boundary.
		if prefix != "/" && len(path) > len(prefix) && path[len(prefix)] != '/' {
			continue
		}

		if !found || len(prefix) > len(best.Prefix) ||
			(len(prefix) == len(best.Prefix) && source.Priority > best.Priority) {
			best = source
			found = true
		}
	}

	if !found {
		return Source{}, false
	}
	return best, true
}
