package catalog

import "strings"

// Source maps a request path to one enabled handler.
type Source struct {
	Name     string
	Prefix   string
	Priority int
	Enabled  bool
}

// Select chooses the best enabled source for path.
func Select(sources []Source, path string) (Source, bool) {
	var best Source
	bestLen := -1
	found := false
	for _, source := range sources {
		if !source.Enabled || source.Prefix == "" {
			continue
		}

		prefix := source.Prefix
		matches := path == prefix
		if !matches && strings.HasPrefix(path, prefix) {
			matches = prefix == "/" || (len(path) > len(prefix) && path[len(prefix)] == '/')
		}
		if !matches {
			continue
		}

		if !found || len(prefix) > bestLen || (len(prefix) == bestLen && source.Priority > best.Priority) {
			best = source
			bestLen = len(prefix)
			found = true
		}
	}
	if !found {
		return Source{}, false
	}
	return best, true
}
