package webhook

import (
	"encoding/json"
	"errors"
	"io"
	"strings"
)

// Event is the signed, durable envelope accepted by the webhook endpoint.
type Event struct {
	ID      string          `json:"id"`
	Type    string          `json:"type"`
	Payload json.RawMessage `json:"payload"`
}

// Validate checks that an event has a usable identity and a JSON object payload.
func (event Event) Validate() error {
	if strings.TrimSpace(event.ID) == "" || strings.TrimSpace(event.Type) == "" {
		return errors.New("event id and type are required")
	}
	var value any
	decoder := json.NewDecoder(strings.NewReader(string(event.Payload)))
	if err := decoder.Decode(&value); err != nil {
		return errors.New("payload must be a JSON object")
	}
	if _, ok := value.(map[string]any); !ok {
		return errors.New("payload must be a JSON object")
	}
	if err := decoder.Decode(new(any)); err != io.EOF {
		return errors.New("payload must contain exactly one JSON value")
	}
	return nil
}
